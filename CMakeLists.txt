cmake_minimum_required(VERSION 3.12)
project(vgraph VERSION 0.1.0 LANGUAGES CXX)
add_library(vgraph src/database.cpp src/vector.cpp src/synthetic.cpp src/hnsw.cpp src/hybrid.cpp)
target_include_directories(vgraph PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_include_directories(vgraph SYSTEM PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/third_party/hnswlib)
find_package(Threads REQUIRED)
target_link_libraries(vgraph PRIVATE Threads::Threads)
target_compile_features(vgraph PUBLIC cxx_std_17)
set_target_properties(vgraph PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(vgraph PRIVATE /W4)
else()
  target_compile_options(vgraph PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(vgraph_demo src/main.cpp)
target_link_libraries(vgraph_demo PRIVATE vgraph)
add_executable(vgraph_search src/search_main.cpp)
target_link_libraries(vgraph_search PRIVATE vgraph)
include(CTest)
if(BUILD_TESTING)
  add_executable(vgraph_tests tests/tests.cpp)
  target_link_libraries(vgraph_tests PRIVATE vgraph)
  add_test(NAME vgraph_tests COMMAND vgraph_tests)
endif()
